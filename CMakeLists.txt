cmake_minimum_required(VERSION 3.20)
project(MyGameEngine LANGUAGES C CXX)
add_compile_definitions(_SECURE_SCL=0)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
add_subdirectory(MyGameEngine/vendor/GLFW)
set(VENDOR ${CMAKE_CURRENT_SOURCE_DIR}/MyGameEngine/vendor)
add_library(Glad STATIC ${VENDOR}/Glad/src/glad.c)
target_include_directories(Glad PUBLIC ${VENDOR}/Glad/include)
add_library(ImGui STATIC ${VENDOR}/imgui/imgui.cpp ${VENDOR}/imgui/imgui_draw.cpp ${VENDOR}/imgui/imgui_tables.cpp ${VENDOR}/imgui/imgui_widgets.cpp ${VENDOR}/imgui/imgui_demo.cpp)
target_include_directories(ImGui PUBLIC ${VENDOR}/imgui)
file(GLOB_RECURSE ENGINE_SOURCES CONFIGURE_DEPENDS MyGameEngine/src/*.cpp)
add_library(MyGameEngine STATIC ${ENGINE_SOURCES} ${VENDOR}/stb_image/stb_image.cpp)
target_precompile_headers(MyGameEngine PRIVATE MyGameEngine/src/MGEpch.h)
target_include_directories(MyGameEngine PUBLIC MyGameEngine/src MyGameEngine/src/MGE ${VENDOR}/spdlog/include ${VENDOR}/glm ${VENDOR}/Math/Mather/include/Mathter ${VENDOR}/stb_image ${VENDOR}/optick/src)
target_compile_definitions(MyGameEngine PUBLIC MGE_PLATFORM_WINDOWS GLFW_INCLUDE_NONE _CRT_SECURE_NO_WARNINGS NOMINMAX "$<$<CONFIG:Debug>:MGE_DEBUG>")
target_link_libraries(MyGameEngine PUBLIC glfw Glad ImGui opengl32)
file(GLOB SANDBOX_SOURCES CONFIGURE_DEPENDS Sandbox/src/*.cpp)
add_executable(Sandbox ${SANDBOX_SOURCES})
target_link_libraries(Sandbox PRIVATE MyGameEngine)
string(TIMESTAMP PERF_BUILD_ID "%Y%m%d-%H%M%S" UTC)
target_compile_definitions(Sandbox PRIVATE PERF_BUILD_ID="${PERF_BUILD_ID}" PERF_BUILD_CONFIG="$<CONFIG>")
set_target_properties(Sandbox PROPERTIES VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/Sandbox")
if(MSVC)
  target_compile_options(MyGameEngine PRIVATE /wd4819)
endif()
